cmake_minimum_required(VERSION 3.24)
project(palace_windows_metis_connection LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Native Windows x64 required")
endif()
set(PW_METIS_PREFIX "" CACHE PATH "Repository-local METIS installation")
set(PW_METIS_SOURCE "" CACHE PATH "Pinned patched METIS source for sort-contract testing")
set(PW_METIS_BUILD "" CACHE PATH "METIS build directory containing generated GKlib headers")
add_executable(metis-connection connection.cpp)
target_compile_features(metis-connection PRIVATE cxx_std_17)
target_include_directories(metis-connection PRIVATE "${PW_METIS_PREFIX}/include")
target_link_libraries(metis-connection PRIVATE "${PW_METIS_PREFIX}/lib/metis.lib")
set_property(TARGET metis-connection PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
if(PW_METIS_SOURCE AND PW_METIS_BUILD)
  add_executable(metis-sort-contract sort_contract.cpp
    "${PW_METIS_SOURCE}/libmetis/gklib_sort.cc")
  target_compile_features(metis-sort-contract PRIVATE cxx_std_11)
  target_compile_definitions(metis-sort-contract PRIVATE WIN32 _CRT_SECURE_NO_DEPRECATE)
  target_include_directories(metis-sort-contract PRIVATE
    "${PW_METIS_SOURCE}/GKlib" "${PW_METIS_SOURCE}/include"
    "${PW_METIS_SOURCE}/libmetis" "${PW_METIS_BUILD}/include")
  set_property(TARGET metis-sort-contract PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
endif()
enable_testing()
add_test(NAME metis-connection COMMAND metis-connection)
set_tests_properties(metis-connection PROPERTIES TIMEOUT 60)
if(TARGET metis-sort-contract)
  add_test(NAME metis-sort-contract COMMAND metis-sort-contract)
  set_tests_properties(metis-sort-contract PROPERTIES TIMEOUT 60)
endif()
