cmake_minimum_required(VERSION 3.24)
project(ceed_windows VERSION 0.12.0 LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_C_COMPILER_ID STREQUAL "IntelLLVM")
  message(FATAL_ERROR "This CPU recipe requires the pinned native IntelLLVM C99 compiler")
endif()
set(CEED_SOURCE "" CACHE PATH "Pinned, patched libCEED source")
file(GLOB interface CONFIGURE_DEPENDS "${CEED_SOURCE}/interface/ceed*.c")
list(FILTER interface EXCLUDE REGEX "ceed-(cuda|hip|jit-source-root-default)\\.c$")
file(GLOB backend CONFIGURE_DEPENDS "${CEED_SOURCE}/backends/ref/*.c" "${CEED_SOURCE}/backends/blocked/*.c" "${CEED_SOURCE}/backends/opt/*.c")
file(GLOB gallery CONFIGURE_DEPENDS "${CEED_SOURCE}/gallery/*/ceed*.c")
add_library(ceed STATIC ${interface} ${backend} ${gallery} windows_alloc.c)
target_compile_features(ceed PRIVATE c_std_11)
set_target_properties(ceed PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_include_directories(ceed PUBLIC "$<BUILD_INTERFACE:${CEED_SOURCE}/include>" "$<INSTALL_INTERFACE:include>" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(ceed PUBLIC CEED_SKIP_VISIBILITY
  "$<$<COMPILE_LANG_AND_ID:CXX,MSVC>:__restrict__=__restrict>" PRIVATE PW_CEED_WINDOWS_CPU
  "CEED_JIT_SOURCE_ROOT_DEFAULT=\"${CMAKE_INSTALL_PREFIX}/include/\""
  "CEED_GIT_VERSION=\"39f259f89332e936122f7e02d6088a1dae3fb628-windows-overlay\""
  "CEED_BUILD_CONFIGURATION=\"Windows x64 static CPU ref/blocked/opt, IntelLLVM C99, DLL CRT\"")
install(TARGETS ceed EXPORT ceed-targets ARCHIVE DESTINATION lib)
install(DIRECTORY "${CEED_SOURCE}/include/" DESTINATION include FILES_MATCHING PATTERN "*.h")
install(EXPORT ceed-targets NAMESPACE ceed:: DESTINATION lib/cmake/ceed)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ceedConfig.cmake" "include(\"\${CMAKE_CURRENT_LIST_DIR}/ceed-targets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ceedConfig.cmake" DESTINATION lib/cmake/ceed)
enable_testing()
add_executable(windows-allocation allocation_test.c)
target_link_libraries(windows-allocation PRIVATE ceed)
add_test(NAME windows-allocation COMMAND windows-allocation)
foreach(test t100-vector t101-vector t200-elemrestriction t500-operator)
  add_executable(${test} "${CEED_SOURCE}/tests/${test}.c")
  target_link_libraries(${test} PRIVATE ceed)
  foreach(backend ref/serial ref/blocked opt/serial opt/blocked)
    string(REPLACE "/" "-" suffix "${backend}")
    add_test(NAME "${test}-${suffix}" COMMAND ${test} "/cpu/self/${backend}")
    # These upstream tests print mismatches but return zero. Require silence.
    set_tests_properties("${test}-${suffix}" PROPERTIES FAIL_REGULAR_EXPRESSION ".+" TIMEOUT 60)
  endforeach()
endforeach()
